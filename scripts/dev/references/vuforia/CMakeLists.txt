cmake_minimum_required(VERSION 3.22)
project(vuforia_reference_contracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VUFORIA_REFERENCE_WITH_SDK "Build the Android SDK version viewer" OFF)
if(VUFORIA_REFERENCE_WITH_SDK)
  if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Vuforia 11.4.4 Android reference requires arm64-v8a")
  endif()
  if(NOT EXISTS "${VUFORIA_SDK_INCLUDE}/VuforiaEngine/VuforiaEngine.h" OR NOT EXISTS "${VUFORIA_SDK_LIBRARY}")
    message(FATAL_ERROR "Supply the official SDK include directory and arm64 libVuforia.so")
  endif()
  add_library(vuforia_sdk SHARED IMPORTED)
  set_target_properties(vuforia_sdk PROPERTIES IMPORTED_LOCATION "${VUFORIA_SDK_LIBRARY}")
  add_library(vuforia_reference SHARED version_probe.cpp)
  target_include_directories(vuforia_reference PRIVATE "${VUFORIA_SDK_INCLUDE}")
  target_link_libraries(vuforia_reference PRIVATE vuforia_sdk)
else()
  enable_testing()
  add_executable(vuforia_public_contracts offline_contracts.cpp)
  target_compile_options(vuforia_public_contracts PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME independent_coordinate_time_contracts COMMAND vuforia_public_contracts)
endif()
