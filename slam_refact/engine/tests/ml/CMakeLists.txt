# ML engine tests (US-016/US-017/RM-1; slam_refact/docs/design/ml-runtime.md §5, §6).
#
# Labels:
#   unit     checked-in data only (data/); passes on a fresh clone.
#   mlmodel  the seven real models from assets/references (gitignored, never copied into the repo)
#            against the numpy reference; needs all seven model files, python3 and numpy.
#            When one is missing configure still succeeds and ONE always-failing test
#            (sr_ml_assets_missing, fail_missing_mlmodel.cmake) is registered, so the label cannot
#            pass silently; -DSLAM_REFACT_ALLOW_BLOCKED_MLMODEL=ON only stops registering that
#            failing test (ml_assets.cmake, RM-1c).
#   mlgate   the 17 tests that judge the regression gate v3 (rm1-regression-gate.v3.md section 9:
#            R1-R3 are read with `-L mlgate`; v3.addendum.json), named in data/mlgate_names.txt.
#            Always together with mlmodel; the face model tests (tests/face) carry mlmodel only.
#            RM-1d: the label is given by name, a missing / stale / extended binary registers a
#            failing test (mlgate_labels.cmake), and the judging command is
#            `ctest --no-tests=error -L '^mlgate$' -E sr_ml_ref_generate` with exactly 17 tests.
#            Not `evidence`: `ctest -L evidence` is a regular expression and the old name
#            `ml-evidence` was selected by it.

include("${CMAKE_CURRENT_SOURCE_DIR}/ml_assets.cmake")

# unit: FlexBuffers, parser robustness, hostile models, single-op cases against the numpy
# reference (checked-in data under data/op_cases, generated by tools/ml_ref/gen_op_cases.py).
sr_add_unit_test(sr_tests_ml_ops
  SOURCES ml_ops_test.cc
  LIBS sr_ml)
target_compile_definitions(sr_tests_ml_ops PRIVATE
  SR_ML_OP_CASES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data/op_cases"
  SR_ML_GATE_TSV="${CMAKE_CURRENT_SOURCE_DIR}/data/model_regression_gate_v3.tsv"
  SR_ML_GATE_AGGREGATE_TSV="${CMAKE_CURRENT_SOURCE_DIR}/data/model_regression_gate_v3_aggregate.tsv")

# mlmodel: needs python3 + numpy for the reference run (a development-only dependency).
set(_sr_ml_models
  engine/resources/face-model.tflite
  engine/resources/face-mesh-model.tflite
  engine/resources/face-ear-model.tflite
  engine/resources/semantics-model.tflite
  official/mediapipe-fork/mediapipe/models/palm_detection.tflite
  official/mediapipe-fork/mediapipe/models/hand_landmark.tflite
  official/mediapipe-fork/mediapipe/models/hand_landmark_3d.tflite)
set(_sr_ml_missing "")
foreach(_m IN LISTS _sr_ml_models)
  if(NOT EXISTS "${SLAM_REFACT_REPO_ROOT}/assets/references/xr_reference/${_m}")
    list(APPEND _sr_ml_missing "${_m}")
  endif()
endforeach()
find_program(SR_ML_PYTHON3 NAMES python3)
# The reference run needs numpy; a python3 without it would only fail later, inside the fixture.
set(_sr_ml_numpy_ok FALSE)
if(SR_ML_PYTHON3)
  execute_process(COMMAND "${SR_ML_PYTHON3}" -I -c "import numpy"
    RESULT_VARIABLE _sr_ml_numpy_rc OUTPUT_QUIET ERROR_QUIET)
  if(_sr_ml_numpy_rc EQUAL 0)
    set(_sr_ml_numpy_ok TRUE)
  endif()
endif()
set(_sr_ml_blocked "")
if(_sr_ml_missing)
  list(JOIN _sr_ml_missing ", " _sr_ml_missing_text)  # no ';' in the text passed to the failing test
  string(APPEND _sr_ml_blocked " missing models: ${_sr_ml_missing_text}.")
endif()
if(NOT SR_ML_PYTHON3)
  string(APPEND _sr_ml_blocked " python3 not found.")
elseif(NOT _sr_ml_numpy_ok)
  string(APPEND _sr_ml_blocked " '${SR_ML_PYTHON3}' cannot import numpy.")
endif()
if(_sr_ml_blocked)
  # RM-1d (review RM-1c MINOR-1): some assets present (a model, or all models without numpy) is a
  # broken checkout; -DSLAM_REFACT_ALLOW_BLOCKED_MLMODEL=ON only covers a fresh clone.
  list(LENGTH _sr_ml_missing _sr_ml_n_missing)
  list(LENGTH _sr_ml_models _sr_ml_n_models)
  set(_sr_ml_partial "")
  if(_sr_ml_n_missing LESS _sr_ml_n_models)
    set(_sr_ml_partial PARTIAL)
  endif()
  sr_ml_report_blocked(sr_ml_assets_missing "mlmodel;mlgate" "${_sr_ml_blocked}" ${_sr_ml_partial})
else()
  set(SR_ML_REF_DIR "${CMAKE_BINARY_DIR}/ml_ref")
  add_test(NAME sr_ml_ref_generate
    COMMAND "${SR_ML_PYTHON3}" -I "${SLAM_REFACT_REPO_ROOT}/slam_refact/tools/ml_ref/ml_ref.py"
            run-all --out "${SR_ML_REF_DIR}")
  set_tests_properties(sr_ml_ref_generate PROPERTIES
    LABELS mlmodel
    FIXTURES_SETUP sr_ml_ref
    TIMEOUT 900)

  add_executable(sr_tests_ml_models ml_models_test.cc)
  target_link_libraries(sr_tests_ml_models PRIVATE sr_ml GTest::gtest_main)
  sr_engine_warnings(sr_tests_ml_models)
  target_compile_definitions(sr_tests_ml_models PRIVATE
    SR_REPO_ROOT="${SLAM_REFACT_REPO_ROOT}"
    SR_ML_REF_DIR="${SR_ML_REF_DIR}"
    SR_ML_GATE_TSV="${CMAKE_CURRENT_SOURCE_DIR}/data/model_regression_gate_v3.tsv"
    SR_ML_GATE_AGGREGATE_TSV="${CMAKE_CURRENT_SOURCE_DIR}/data/model_regression_gate_v3_aggregate.tsv")
  include(GoogleTest)
  gtest_discover_tests(sr_tests_ml_models
    PROPERTIES LABELS mlmodel FIXTURES_REQUIRED sr_ml_ref TIMEOUT 900
    DISCOVERY_TIMEOUT 60)
  # The 17 tests that judge the regression gate v3 carry `mlgate` besides `mlmodel` (RM-1c), chosen
  # by NAME from data/mlgate_names.txt (RM-1d, review RM-1c MAJOR-1), not "every test of the
  # binary": gtest_discover_tests cannot pass a two-element LABELS list through its PROPERTIES
  # argument (the ';' splits it into a property name), so an include file that ctest reads after the
  # discovered tests exist (<target>_TESTS holds their names) adds the labels. Binary not built,
  # stale (fewer / other tests) or extended: it registers ONE failing test under `mlmodel;mlgate`
  # (mlgate_labels.cmake, fail_gate_names.cmake).
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sr_tests_ml_models_gate_labels.cmake"
    "set(SR_ML_GATE_NAMES_FILE \"${CMAKE_CURRENT_SOURCE_DIR}/data/mlgate_names.txt\")\n"
    "set(SR_ML_GATE_FAIL_SCRIPT \"${CMAKE_CURRENT_SOURCE_DIR}/fail_gate_names.cmake\")\n"
    "set(SR_ML_GATE_CMAKE \"${CMAKE_COMMAND}\")\n"
    "if(DEFINED sr_tests_ml_models_TESTS)\n"
    "  set(SR_ML_GATE_TESTS_DEFINED TRUE)\n"
    "  set(SR_ML_GATE_TESTS \"\${sr_tests_ml_models_TESTS}\")\n"
    "else()\n"
    "  set(SR_ML_GATE_TESTS_DEFINED FALSE)\n"
    "endif()\n"
    "include(\"${CMAKE_CURRENT_SOURCE_DIR}/mlgate_labels.cmake\")\n")
  set_property(DIRECTORY APPEND PROPERTY TEST_INCLUDE_FILES
    "${CMAKE_CURRENT_BINARY_DIR}/sr_tests_ml_models_gate_labels.cmake")
endif()
